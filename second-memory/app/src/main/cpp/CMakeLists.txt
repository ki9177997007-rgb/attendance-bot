cmake_minimum_required(VERSION 3.22.1)
project(memory_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../../../../vendor/llama.cpp llama-build)
if(ANDROID)
 add_library(memory_ai SHARED memory_ai.cpp)
 target_link_libraries(memory_ai llama android log)
 target_link_options(memory_ai PRIVATE "-Wl,-z,max-page-size=16384")
else()
 add_executable(memory_ai memory_ai.cpp)
 target_link_libraries(memory_ai llama)
endif()
